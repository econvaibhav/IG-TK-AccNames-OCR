\documentclass[11pt,border=12pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc}
\begin{document}
\hyphenpenalty=10000
\exhyphenpenalty=10000
\begin{tikzpicture}[
  x=1cm,y=1cm,
  box/.style={draw=black,line width=.45pt,fill=white,align=center,inner sep=8pt,text width=7.4cm},
  flow/.style={-{Stealth[length=2mm,width=1.4mm]},line width=.5pt},
  note/.style={font=\fontsize{8}{10}\selectfont,align=center},
  every node/.style={text=black,font=\fontsize{9}{11}\selectfont}
]
\node[box,text width=10.7cm] (input) at (4.7,0.4) {
  \textbf{One clip or a folder of clips}\\
  Process each video in turn; collect all results in one workbook.\\
  Select the Instagram, TikTok or custom crop layout.};
\node[box] (crop) at (0,-2.7) {
  \textbf{Locate the account region}\\
  Resize each sampled frame to $540\times960$.\\[4pt]
  Instagram: profile search strip $\rightarrow$ lowest circle\\
  $\rightarrow$ adjacent $380\times40$ account crop.\\[4pt]
  TikTok: fixed lower-left region, or a custom ROI.};
\node[box] (reader) at (0,-6.8) {
  \textbf{Choose the reader and languages}\\
  \texttt{-{}-lang europe} or \texttt{-{}-lang en bg}\\[4pt]
  Paddle: 32 language options, three readers.\\
  Latin v6; Cyrillic and Greek v5.\\[4pt]
  English, Bulgarian, Croatian, French, Hungarian,\\
  Finnish, Swedish, German, Polish, Spanish,\\
  Portuguese and other European languages.\\
  See \texttt{-{}-list-languages} for codes.\\[4pt]
  Read text and confidence; remove interface labels.};
\node[box] (votes) at (0,-10.8) {
  \textbf{Read forward and backward}\\
  Sample frames in each direction.\\
  Count one vote per name in each frame;\\
  stop a pass when the required support is reached.\\[4pt]
  Compare candidates and flag uncertainty.};
\node[box,minimum height=4.85cm] (frames) at (9.4,-4.6) {
  \textbf{Keep six moments as review evidence}\\[9pt]
  \begin{tabular}{@{}ll@{}}
    Fixed times & $1\,\mathrm{s},\;3\,\mathrm{s},\;T-1\,\mathrm{s}$\\[8pt]
    Quarter & $0.25T$\\[4pt]
    Median & $0.50T$\\[4pt]
    Three quarters & $0.75T$
  \end{tabular}\\[11pt]
  Save the account crop and the full frame.\\
  Enlarge images to inspect the visible name.\\[5pt]
  {\fontsize{8}{10}\selectfont $T$ is clip duration. Unavailable fixed times are marked.\\
  Requested times can select the same frame.}};
\node[box,minimum height=2.25cm] (review) at (9.4,-10.8) {
  \textbf{Check the name in the browser}\\
  Tick \emph{Mark as reviewed}; keep a correct name unchanged.\\
  Edit the name only when a correction is needed.\\
  Tick \emph{Cannot read} when the evidence is unclear.\\[4pt]
  Add an optional note; press \emph{Save changes}.};
\node[box] (original) at (0,-14.4) {
  \textbf{Preserve the original readings}\\
  Keep raw text, confidence, vote counts,\\
  frame positions, engine and run settings.\\[4pt]
  Agreement between passes is supporting evidence,\\
  not verification of the account's identity.};
\node[box] (state) at (9.4,-14.4) {
  \textbf{Store the review decision}\\
  Infer \emph{confirmed} or \emph{corrected} from the name;\\
  keep unreadable clips unresolved.\\[4pt]
  Retain notes and correction history.\\
  Reopening the review restores saved decisions.};
\node[box,text width=10.7cm] (excel) at (4.7,-17.85) {
  \textbf{Update and download Excel}\\
  \texttt{reviewed.xlsx}: one row per clip, original readings alongside\\
  \texttt{final\_names}, \texttt{review\_status} and reviewer notes.\\[4pt]
  Saving a review refreshes the workbook for the whole run.};
\draw[flow] (input.south) -- ++(0,-.38) -| (crop.north);
\draw[flow] (input.south) -- ++(0,-.38) -| (frames.north);
\draw[flow] (crop) -- (reader);
\draw[flow] (reader) -- (votes);
\draw[flow] (votes) -- (original);
\draw[flow] (votes.east) -- (review.west);
\draw[flow] (frames) -- (review);
\draw[flow] (review) -- (state);
\draw[flow] (original.south) -- ++(0,-.45) -| ($(excel.north)+(-2.6,0)$);
\draw[flow] (state.south) -- ++(0,-.45) -| ($(excel.north)+(2.6,0)$);
\node[note,text width=15.3cm] at (4.7,-19.9) {
  Example: \texttt{thestoryofourhome-pl} $\rightarrow$ \texttt{thestoryofourhome.pl}; a punctuation correction.\\[3pt]
  Review $\rightarrow$ Excel is one-way. Editing Excel does not change the review.\\
  Corrections update the saved result; they do not retrain the OCR reader.};
\end{tikzpicture}
\end{document}
