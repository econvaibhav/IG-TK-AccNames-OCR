\node[box] (votes) at (3.4,-7.0) {\textbf{Read in both directions}\\Sample forward and backward through the clip.\\Filter interface text; count one vote per name per frame.\\Stop a pass at the required vote support.};
\node[box] (compare) at (3.4,-9.6) {\textbf{Compare the two passes}\\Keep names, vote counts and review flags.\\Both passes can agree and still share an error.};
\node[box,text width=7.2cm,minimum height=5.1cm] (evidence) at (13.6,-4.8) {\textbf{Save six moments for human review}\\[10pt]

\end{tabular}\\[12pt]
Show an account crop and the full frame at each time.\\Click a frame to inspect it at a larger size.\\[5pt]
{\fontsize{8}{10}\selectfont Frame index $i=\min(N-1,\,\max(0,\,\mathrm{round}(tf)))$.\\Out-of-range fixed times are marked unavailable.}};
\node[box,text width=7.2cm,minimum height=2.15cm] (review) at (13.6,-9.6) {\textbf{Review in the local browser}\\Compare the readings against the visible handle.\\Confirm, correct, or mark the name unreadable.\\Add a note and press \emph{Save correction}.};
\node[box] (raw) at (3.4,-12.6) {\textbf{Original evidence remains available}\\\texttt{results.csv / results.xlsx}\\\texttt{details.jsonl}: raw text, scores and frames\\\texttt{run.json}: engine, settings and versions};
\node[box,text width=7.2cm] (saved) at (13.6,-12.6) {\textbf{Save the reviewed decision}\\\texttt{review\_state.json} retains the account names,\\decision, note, revision and correction history.\\Reopening the review restores saved edits.};
\node[box,text width=7.2cm] (exports) at (13.6,-15.35) {\textbf{Refresh Excel and CSV}\\\texttt{reviewed.xlsx}\quad\texttt{reviewed.csv}\\Read \texttt{final\_names} and \texttt{review\_status}.\\Unreviewed or unreadable clips stay unresolved.};
\node[box,text width=7.2cm] (example) at (3.4,-15.35) {\textbf{One punctuation correction}\\OCR: \texttt{thestoryofourhome-pl}\\Visible handle: \texttt{thestoryofourhome.pl}\\Saved decision: \texttt{corrected}};
